```go
// domain/models/my_feature.go
package models

import (
    "time"
    "gorm.io/gorm"
)

type MyFeature struct {
    ID        uint           `gorm:"primaryKey"`
    Name      string         `gorm:"type:varchar(100);not null"`
    Status    string         `gorm:"type:varchar(20);default:'active'"`
    Metadata  string         `gorm:"type:text"`
    CreatedAt time.Time      `gorm:"autoCreateTime"`
    UpdatedAt time.Time      `gorm:"autoUpdateTime"`
    DeletedAt gorm.DeletedAt `gorm:"index"`
}

func (m *MyFeature) TableName() string {
    return "my_features"
}
```

#### 1.2 创建 Service

```go
// application/services/my_feature_service.go
package services

import (
    "github.com/drama-generator/backend/domain/models"
    "github.com/drama-generator/backend/pkg/logger"
    "gorm.io/gorm"
)

type MyFeatureService struct {
    db  *gorm.DB
    log *logger.Logger
}

func NewMyFeatureService(db *gorm.DB, log *logger.Logger) *MyFeatureService {
    return &MyFeatureService{
        db:  db,
        log: log,
    }
}

func (s *MyFeatureService) Create(name string) (*models.MyFeature, error) {
    feature := &models.MyFeature{
        Name:   name,
        Status: "active",
    }

    if err := s.db.Create(feature).Error; err != nil {
        s.log.Errorw("Failed to create feature", "error", err)
        return nil, err
    }

    return feature, nil
}

func (s *MyFeatureService) GetByID(id uint) (*models.MyFeature, error) {
    var feature models.MyFeature
    if err := s.db.First(&feature, id).Error; err != nil {
        return nil, err
    }
    return &feature, nil
}

func (s *MyFeatureService) List(page, pageSize int) ([]models.MyFeature, int64, error) {
    var features []models.MyFeature
    var total int64

    query := s.db.Model(&models.MyFeature{})
    query.Count(&total)

    if err := query.Offset((page - 1) * pageSize).Limit(pageSize).Find(&features).Error; err != nil {
        return nil, 0, err
    }

    return features, total, nil
}
```

#### 1.3 创建 Handler

```go
// api/handlers/my_feature_handler.go
package handlers

import (
    "strconv"

    "github.com/drama-generator/backend/application/services"
    "github.com/drama-generator/backend/pkg/config"
    "github.com/drama-generator/backend/pkg/logger"
    "github.com/gin-gonic/gin"
    "gorm.io/gorm"
)

type MyFeatureHandler struct {
    db  *gorm.DB
    cfg *config.Config
    log *logger.Logger
    svc *services.MyFeatureService
}

func NewMyFeatureHandler(db *gorm.DB, cfg *config.Config, log *logger.Logger) *MyFeatureHandler {
    return &MyFeatureHandler{
        db:  db,
        cfg: cfg,
        log: log,
        svc: services.NewMyFeatureService(db, log),
    }
}

func (h *MyFeatureHandler) Create(c *gin.Context) {
    var req struct {
        Name string `json:"name" binding:"required"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }

    feature, err := h.svc.Create(req.Name)
    if err != nil {
        c.JSON(500, gin.H{"error": "Failed to create"})
        return
    }

    c.JSON(200, feature)
}

func (h *MyFeatureHandler) Get(c *gin.Context) {
    id, _ := strconv.ParseUint(c.Param("id"), 10, 32)

    feature, err := h.svc.GetByID(uint(id))
    if err != nil {
        c.JSON(404, gin.H{"error": "Not found"})
        return
    }

    c.JSON(200, feature)
}

func (h *MyFeatureHandler) List(c *gin.Context) {
    page, _ := strconv.Atoi(c.DefaultQuery("page", "1"))
    pageSize, _ := strconv.Atoi(c.DefaultQuery("page_size", "20"))

    features, total, err := h.svc.List(page, pageSize)
    if err != nil {
        c.JSON(500, gin.H{"error": "Failed to list"})
        return
    }

    c.JSON(200, gin.H{
        "data":  features,
        "total": total,
        "page":  page,
        "size":  pageSize,
    })
}
```

```go
func (h *MyHandler) CallAI(c *gin.Context) {
    var req struct {
        Prompt string `json:"prompt" binding:"required"`
    }
    c.ShouldBindJSON(&req)

    // 获取 AI 配置
    var aiConfig models.AIConfig
    h.db.Where("provider = ? AND enabled = ?", "openai", true).First(&aiConfig)

    // 调用 AI
    client := ai.GetClient(aiConfig.Provider, aiConfig.APIKey)
    resp, err := client.Chat(c.Request.Context(), &ai.ChatRequest{
        Model:    aiConfig.Model,
        Messages: []ai.Message{{Role: "user", Content: req.Prompt}},
    })

    if err != nil {
        c.JSON(500, gin.H{"error": err.Error()})
        return
    }

    c.JSON(200, gin.H{"result": resp.Content})
}
```

## 3. 异步任务处理

```go
// 创建异步任务
func (h *MyHandler) CreateAsyncTask(c *gin.Context) {
    // 创建任务记录
    task := &models.Task{
        Type:        "image_generation",
        ResourceID:  1,
        ResourceType: "character",
        Status:      "pending",
        Progress:    0,
    }
    h.db.Create(task)

    // 异步执行
    go func() {
        h.db.Model(task).Updates(map[string]interface{}{
            "status":   "processing",
            "started_at": time.Now(),
        })

        // 执行耗时操作
        result, err := doHeavyWork()

        if err != nil {
            h.db.Model(task).Updates(map[string]interface{}{
                "status":  "failed",
                "error":  err.Error(),
            })
            return
        }

        h.db.Model(task).Updates(map[string]interface{}{
            "status":      "completed",
            "progress":    100,
            "result":      result,
            "completed_at": time.Now(),
        })
    }()

    c.JSON(200, gin.H{"task_id": task.ID})
}

// 查询任务状态
func (h *MyHandler) GetTaskStatus(c *gin.Context) {
    id, _ := strconv.ParseUint(c.Param("task_id"), 10, 32)

    var task models.Task
    if err := h.db.First(&task, id).Error; err != nil {
        c.JSON(404, gin.H{"error": "Task not found"})
        return
    }

    c.JSON(200, task)
}
```

## 4. 文件上传处理

```go
// api/handlers/upload_handler.go
func (h *UploadHandler) UploadImage(c *gin.Context) {
    file, err := c.FormFile("file")
    if err != nil {
        c.JSON(400, gin.H{"error": "No file uploaded"})
        return
    }

    // 生成唯一文件名
    ext := filepath.Ext(file.Filename)
    filename := fmt.Sprintf("%d%s", time.Now().Unix(), ext)

    // 保存文件
    filepath := filepath.Join(h.cfg.Storage.LocalPath, filename)
    if err := c.SaveUploadedFile(file, filepath); err != nil {
        c.JSON(500, gin.H{"error": "Failed to save file"})
        return
    }

    // 返回访问 URL
    url := fmt.Sprintf("%s/%s", h.cfg.Storage.BaseURL, filename)
    c.JSON(200, gin.H{"url": url})
}
```

## 5. 数据库事务

```go
// 事务示例
func (s *MyService) CreateWithRelations(drama *models.Drama, episodes []models.Episode) error {
    return s.db.Transaction(func(tx *gorm.DB) error {
        // 创建主记录
        if err := tx.Create(drama).Error; err != nil {
            return err
        }

        // 批量创建关联记录
        for i := range episodes {
            episodes[i].DramaID = drama.ID
        }
        if err := tx.Create(&episodes).Error; err != nil {
            return err
        }

        return nil
    })
}
```
