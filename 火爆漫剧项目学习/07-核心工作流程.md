# 好剧生成器 核心工作流程

## 完整工作流程

```
┌─────────────────────────────────────────────────────────────────────────┐
│                         短剧创作全流程                                   │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  1. 创建项目                                                            │
│     └── 输入剧集标题、类型、风格                                          │
│                                                                         │
│  2. 上传剧本                                                            │
│     └── 导入剧本文本                                                     │
│                                                                         │
│  3. AI 提取角色                                                         │
│     └── 分析剧本 → 提取角色列表 (姓名/描述/外貌)                          │
│                                                                         │
│  4. AI 生成角色图片                                                     │
│     └── 选择角色 → 生成图片 → 可编辑/重新生成                             │
│                                                                         │
│  5. AI 提取场景                                                         │
│     └── 分析剧本 → 提取场景列表 (地点/时间/描述)                          │
│                                                                         │
│  6. AI 生成场景图片                                                     │
│     └── 选择场景 → 生成图片 → 可编辑/重新生成                             │
│                                                                         │
│  7. AI 生成分镜                                                         │
│     └── 分析场景 → 生成镜头脚本 (分镜/镜头/动作/对话)                      │
│                                                                         │
│  8. AI 生成视频                                                         │
│     └── 选择分镜 → 生成视频片段 → 可编辑/重新生成                         │
│                                                                         │
│  9. 视频合成                                                            │
│     └── 调整顺序 → FFmpeg 合并 → 最终视频                                 │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

## 1. 角色提取流程

```go
// application/services/script_generation_service.go
func (s *ScriptGenerationService) ExtractCharacters(scriptContent string, cfg *models.AIConfig) ([]Character, error) {
    // 构建 Prompt
    prompt := fmt.Sprintf(`
从以下剧本中提取所有角色信息，包括：
1. 角色姓名
2. 角色类型（主角/配角/反派/客串）
3. 外貌描述
4. 性格特点
5. 声音风格

剧本内容：
%s

请以 JSON 数组格式返回每个角色的详细信息。
    `, scriptContent)

    // 调用 AI
    client := ai.GetClient(cfg.Provider, cfg.APIKey)
    resp, err := client.Chat(context.Background(), &ai.ChatRequest{
        Model:    cfg.Model,
        Messages: []ai.Message{{Role: "user", Content: prompt}},
    })

    // 解析 JSON
    var characters []Character
    json.Unmarshal([]byte(resp.Content), &characters)

    return characters, nil
}
```

## 2. 角色图片生成流程

```go
// application/services/image_generation_service.go
func (s *ImageGenerationService) GenerateCharacterImage(
    character *models.Character,
    cfg *models.AIConfig,
) (*models.ImageGeneration, error) {

    // 构建 Prompt
    prompt := fmt.Sprintf(
        "A realistic portrait of %s, %s, %s, high quality, 4k",
        character.Name,
        getOrEmpty(character.Description),
        getOrEmpty(character.Appearance),
    )

    // 调用图像生成服务
    imageClient := image.GetClient(cfg.Provider, cfg.APIKey)
    resp, err := imageClient.GenerateImage(context.Background(), &image.ImageGenerateRequest{
        Model:  cfg.Model,
        Prompt: prompt,
        Width:  512,
        Height: 768,
    })

    // 保存记录
    imgGen := &models.ImageGeneration{
        DramaID:      character.DramaID,
        CharacterID:  &character.ID,
        Type:         "character",
        Prompt:       prompt,
        Model:        cfg.Model,
        Status:       resp.Status,
        ImageURL:     resp.ImageURL,
        LocalPath:    resp.LocalPath,
    }

    s.db.Create(imgGen)

    // 异步获取结果
    go s.waitForImageCompletion(imgGen, imageClient, resp.TaskID)

    return imgGen, nil
}
```

## 5. 视频生成

```go
// application/services/video_generation_service.go
func (s *VideoGenerationService) GenerateVideo(
    storyboardID uint,
    cfg *models.AIConfig,
) (*VideoGeneration, error) {

    // 获取分镜信息
    var storyboard models.Storyboard
    s.db.Preload("Background").First(&storyboard, storyboardID)

    // 构建视频生成请求
    req := &video.VideoGenerateRequest{
        Model:    cfg.Model,
        Prompt:   *storyboard.VideoPrompt,
        ImageURL: *storyboard.Background.ImageURL,
        Duration: storyboard.Duration,
    }

    // 调用视频生成服务
    videoClient := video.GetClient(cfg.Provider, cfg.APIKey)
    resp, err := videoClient.GenerateVideo(context.Background(), req)

    // 保存记录
    videoGen := &models.VideoGeneration{
        StoryboardID: storyboardID,
        EpisodeID:    &storyboard.EpisodeID,
        DramaID:      storyboard.Episode.DramaID,
        Prompt:       req.Prompt,
        Model:        req.Model,
        Status:       "processing",
        TaskID:       &resp.TaskID,
    }

    s.db.Create(videoGen)

    // 异步等待完成
    go s.waitForVideoCompletion(videoGen, videoClient, resp.TaskID)

    return videoGen, nil
}
```

## 6. 视频合并

```go
// application/services/video_merge_service.go
func (s *VideoMergeService) MergeVideos(
    videoIDs []uint,
    cfg *VideoMergeConfig,
) (*VideoMerge, error) {

    // 获取所有视频
    var videos []models.VideoGeneration
    s.db.Find(&videos, videoIDs)

    // 准备 FFmpeg 输入
    inputFiles := make([]string, len(videos))
    for i, v := range videos {
        inputFiles[i] = *v.LocalPath
    }

    // 生成输出路径
    outputFile := filepath.Join(cfg.OutputDir, fmt.Sprintf("merge_%d.mp4", time.Now().Unix()))

    // 调用 FFmpeg 合并
    err := ffmpeg.MergeVideos(inputFiles, outputFile)
    if err != nil {
        return nil, err
    }

    // 保存合并记录
    merge := &models.VideoMerge{
        DramaID:    videos[0].DramaID,
        EpisodeID: videos[0].EpisodeID,
        VideoURL:  &outputFile,
        Duration:  calculateTotalDuration(videos),
        Status:    "completed",
    }

    s.db.Create(merge)

    return merge, nil
}
```

### FFmpeg 合并命令

```bash
ffmpeg -i input1.mp4 -i input2.mp4 -filter_complex "[0:v][1:v]concat=n=2:v=1:a=1[v]" -map "[v]" output.mp4
```

```go
// application/services/task_service.go
func (s *TaskService) CreateTask(taskType string, resourceID uint, resourceType string) *models.Task {
    task := &models.Task{
        Type:         taskType,
        ResourceID:    resourceID,
        ResourceType: resourceType,
        Status:       "pending",
        Progress:     0,
    }

    s.db.Create(task)
    return task
}

func (s *TaskService) UpdateTaskProgress(taskID uint, progress int, status string) {
    s.db.Model(&models.Task{}).Where("id = ?", taskID).Updates(map[string]interface{}{
        "progress": progress,
        "status":   status,
    })
}
```
