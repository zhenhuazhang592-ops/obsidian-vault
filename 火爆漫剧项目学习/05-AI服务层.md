```go
// pkg/ai/client.go
type AIClient interface {
    // 聊天
    Chat(ctx context.Context, req *ChatRequest) (*ChatResponse, error)

    // 流式聊天
    StreamChat(ctx context.Context, req *ChatRequest) (*ChatStreamReader, error)

    // 获取模型列表
    ListModels(ctx context.Context) ([]Model, error)
}

type ChatRequest struct {
    Model       string
    Messages    []Message
    Temperature float64
    MaxTokens   int
}

type Message struct {
    Role    string // "system" | "user" | "assistant"
    Content string
}
```

### OpenAI 客户端实现

```go
// pkg/ai/openai_client.go
type OpenAIClient struct {
    apiKey  string
    baseURL string
    client  *http.Client
}

func (c *OpenAIClient) Chat(ctx context.Context, req *ai.ChatRequest) (*ai.ChatResponse, error) {
    // 构建请求
    payload := map[string]interface{}{
        "model":       req.Model,
        "messages":    req.Messages,
        "temperature": req.Temperature,
        "max_tokens":  req.MaxTokens,
    }

    // 调用 API
    resp, err := c.doRequest(ctx, "POST", "/chat/completions", payload)
    if err != nil {
        return nil, err
    }

    // 解析响应
    var result map[string]interface{}
    json.Unmarshal(resp, &result)

    choices := result["choices"].([]interface{})
    message := choices[0].(map[string]interface{})["message"].(map[string]interface{})

    return &ai.ChatResponse{
        Content: message["content"].(string),
        Model:   result["model"].(string),
    }, nil
}
```

```go
// application/services/ai_service.go
type AIService struct {
    db  *gorm.DB
    log *logger.Logger
}

func (s *AIService) GetClient(configType string) (ai.AIClient, error) {
    var config models.AIConfig
    if err := s.db.Where("provider = ? AND enabled = ?", configType, true).First(&config).Error; err != nil {
        return nil, err
    }

    switch config.Provider {
    case "openai":
        return openai.NewClient(config.APIKey, config.BaseURL), nil
    case "google":
        return gemini.NewClient(config.APIKey), nil
    default:
        return nil, errors.New("unsupported provider")
    }
}
```

## 5. Prompt i18n 支持

```go
// application/services/prompt_i18n.go
type PromptI18n struct {
    translations map[string]map[string]string
}

func (p *PromptI18n) Translate(key string, lang string) string {
    if trans, ok := p.translations[lang]; ok {
        if val, ok := trans[key]; ok {
            return val
        }
    }
    // 默认英文
    if trans, ok := p.translations["en"]; ok {
        return trans[key]
    }
    return key
}

func (p *PromptI18n) BuildPrompt(template string, params map[string]string, lang string) string {
    result := template
    for k, v := range params {
        result = strings.ReplaceAll(result, "{"+k+"}", v)
    }
    return p.Translate(result, lang)
}
```
