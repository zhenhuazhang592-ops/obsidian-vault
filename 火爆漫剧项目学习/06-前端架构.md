## 1. API 请求封装

```typescript
// src/utils/request.ts
import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios'

const request: AxiosInstance = axios.create({
  baseURL: '/api/v1',
  timeout: 60000,
})

// 请求拦截器
request.interceptors.request.use(
  (config) => {
    // 添加 token
    const token = localStorage.getItem('token')
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
request.interceptors.response.use(
  (response: AxiosResponse) => {
    return response.data
  },
  (error) => {
    // 统一错误处理
    const message = error.response?.data?.error || '请求失败'
    ElMessage.error(message)
    return Promise.reject(error)
  }
)

export default request
```

## 2. API 模块

```typescript
// src/api/drama.ts
import request from '@/utils/request'
import type { Drama, CreateDramaRequest, UpdateDramaRequest } from '@/types/drama'

export function getDramas(params: {
  page?: number
  page_size?: number
  status?: string
  keyword?: string
}) {
  return request.get<{ data: Drama[]; total: number }>('/dramas', { params })
}

export function getDrama(id: number) {
  return request.get<Drama>(`/dramas/${id}`)
}

export function createDrama(data: CreateDramaRequest) {
  return request.post<Drama>('/dramas', data)
}

export function updateDrama(id: number, data: UpdateDramaRequest) {
  return request.put<Drama>(`/dramas/${id}`, data)
}

export function deleteDrama(id: number) {
  return request.delete(`/dramas/${id}`)
}
```

## 4. 状态管理

```typescript
// src/stores/episode.ts
import { defineStore } from 'pinia'
import type { Episode } from '@/types/drama'

interface EpisodeState {
  currentEpisode: Episode | null
  episodes: Episode[]
  loading: boolean
}

export const useEpisodeStore = defineStore('episode', {
  state: (): EpisodeState => ({
    currentEpisode: null,
    episodes: [],
    loading: false,
  }),

  actions: {
    async fetchEpisodes(dramaId: number) {
      this.loading = true
      try {
        const res = await getEpisodes(dramaId)
        this.episodes = res.data
      } finally {
        this.loading = false
      }
    },

    setCurrentEpisode(episode: Episode) {
      this.currentEpisode = episode
    },
  },
})
```

## 7. 视频处理

### FFmpeg WebAssembly

```typescript
// src/utils/ffmpeg.ts
import { FFmpeg } from '@ffmpeg/ffmpeg'
import { fetchFile, toBlobURL } from '@ffmpeg/util'

let ffmpeg: FFmpeg | null = null

export async function initFFmpeg() {
  if (ffmpeg) return ffmpeg

  ffmpeg = new FFmpeg()
  const baseURL = '/ffmpeg'

  await ffmpeg.load({
    coreURL: await toBlobURL(`${baseURL}/ffmpeg-core.js`, 'text/javascript'),
    wasmURL: await toBlobURL(`${baseURL}/ffmpeg-core.wasm`, 'application/wasm'),
  })

  return ffmpeg
}

export async function mergeVideos(files: File[]): Promise<Blob> {
  const ff = await initFFmpeg()

  // 写入输入文件
  for (let i = 0; i < files.length; i++) {
    await ff.writeFile(`input${i}.mp4`, await fetchFile(files[i]))
  }

  // 执行合并
  await ff.exec([
    '-i', 'input0.mp4',
    '-i', 'input1.mp4',
    '-filter_complex', 'concat=n=2:v=1:a=1',
    'output.mp4',
  ])

  // 读取输出
  const data = await ff.readFile('output.mp4')
  return new Blob([data], { type: 'video/mp4' })
}
```
