## 4. 剧本解析实现

### 解析流程

```typescript
// src/lib/script/script-parser.ts
async function parseScript(text: string) {
  // 1. 场景分割
  const scenes = splitScenes(text);

  // 2. 分镜拆分
  const shots = scenes.map(s => splitShots(s));

  // 3. 对白提取
  const dialogues = extractDialogues(shots);

  // 4. 角色识别
  const characters = identifyCharacters(dialogues);

  // 5. 情绪标注
  const emotions = annotateEmotions(shots);

  return { scenes, shots, characters, emotions };
}
```

## 5. 分镜提示词构建

### 三层融合

```typescript
// src/components/panels/sclass/sclass-prompt-builder.ts
function buildPrompt(shot: Shot): string {
  // 第一层：动作描述
  const action = shot.action;

  // 第二层：镜头语言
  const cinematography = `${shot.shotType}, ${shot.angle}, ${shot.movement}`;

  // 第三层：对白唇形同步
  const dialogue = shot.dialogue ? `[Lip-sync] ${shot.dialogue}` : '';

  // 融合
  return `${action}\n${cinematography}\n${dialogue}`;
}
```

## 6. 任务队列系统

### 多 Key 轮询

```typescript
// src/packages/ai-core/api/task-queue.ts
class TaskQueue {
  private keys: string[];

  async submitTask(task: Task) {
    // 轮询选择可用 Key
    const key = this.selectKey();
    return this.executeWithKey(task, key);
  }

  private selectKey(): string {
    // 简单轮询或负载均衡
    const index = this.currentIndex % this.keys.length;
    return this.keys[index];
  }
}
```

### 自动重试

```typescript
// src/lib/utils/retry.ts
async function withRetry<T>(
  fn: () => Promise<T>,
  options: { maxRetries: number; delay: number }
): Promise<T> {
  for (let i = 0; i < options.maxRetries; i++) {
    try {
      return await fn();
    } catch (error) {
      if (i === options.maxRetries - 1) throw error;
      await delay(options.delay);
    }
  }
}
```
