# 魔因漫创 数据类型定义

## 核心数据类型

### 项目 (TProject)

```typescript
// src/types/project.ts
interface TProject {
  id: string;
  name: string;
  thumbnail: string;
  createdAt: Date;
  updatedAt: Date;
  scenes: Scene[];
  currentSceneId: string;
  mediaItems?: string[];
  backgroundColor?: string;
  backgroundType?: "color" | "blur";
  blurIntensity?: BlurIntensity;
  fps?: number;
  bookmarks?: number[];
  canvasSize: CanvasSize;
  canvasMode: "preset" | "original" | "custom";
}

type BlurIntensity = 4 | 8 | 18;

interface Scene {
  id: string;
  name: string;
  isMain: boolean;
  createdAt: Date;
  updatedAt: Date;
}
```

### 剧本 (Script)

```typescript
// src/types/script.ts
interface Script {
  id: string;
  projectId: string;
  title: string;
  content: string;
  scenes: ScriptScene[];
  characters: Character[];
  createdAt: Date;
  updatedAt: Date;
}

interface ScriptScene {
  id: string;
  index: number;
  title: string;
  description: string;
  location: string;
  time: string;
  shots: Shot[];
}

interface Shot {
  id: string;
  sceneId: string;
  index: number;
  shotType: ShotType;       // ECU/CU/MS/WS/EWS
  angle: Angle;             // 俯视/仰视/平视
  movement: Movement;      // 固定/推/拉/摇/移
  action: string;
  dialogue?: string;
  characters: string[];    // 出现的角色
}

type ShotType = 'ECU' | 'CU' | 'MS' | 'WS' | 'EWS';
type Angle = 'high' | 'low' | 'eye';
type Movement = 'fixed' | 'push' | 'pull' | 'pan' | 'tilt' | 'dolly';
```

### 角色 (Character)

```typescript
interface Character {
  id: string;
  name: string;
  role: 'protagonist' | 'antagonist' | 'supporting' | 'cameo';
  description: string;         // 角色描述
  appearance: string;          // 外貌特征
  personality: string;         // 性格特点
  referenceImages: string[];   // 参考图
  seedValue?: string;         // 种子值
  bible?: CharacterBible;      // 角色圣经
}

interface CharacterBible {
  identity: string;            // 身份锚点
  appearance: string;          // 外貌锚点
  costume: string;            // 服装锚点
  expression: string;         // 表情锚点
  action: string;             // 动作锚点
}
```

### 媒体 (Media)

```typescript
// src/types/media.ts
interface MediaItem {

  url: string;
  localPath?: string;
  thumbnail?: string;
  width?: number;
  height?: number;
  duration?: number;
  createdAt: Date;
}
```

## 状态管理类型

### API 配置

```typescript
// src/stores/api-config-store.ts
interface APIConfig {
  providers: AIProvider[];
  mappings: ServiceMapping[];
}

interface AIProvider {
  id: string;
  name: string;
  type: 'memefast' | 'runninghub' | string;
  apiKeys: string[];
  enabled: boolean;
}

interface ServiceMapping {
  service: 'text-to-image' | 'image-to-video' | 'text-to-video';
  providerId: string;
  model: string;
}
```

### S级状态

```typescript
// src/stores/sclass-store.ts
interface SClassState {
  shots: SClassShot[];
  groups: ShotGroup[];
  generationStatus: GenerationStatus;
}

interface SClassShot {
  id: string;
  shotId: string;
  firstFramePrompt: string;
  lastFramePrompt: string;
  videoPrompt: string;
  imageRef?: string;        // 参考图
  videoRef?: string;        // 参考视频
  audioRef?: string;        // 参考音频
  duration?: number;
}

interface ShotGroup {
  id: string;
  shotIds: string[];
  mergedPrompt: string;
  status: 'pending' | 'generating' | 'completed' | 'failed';
  result?: string;
}
```
