## 二、文本生成服务（AIText）

### 2.1 服务封装

```typescript
// src/utils/ai/text/index.ts
class AITextService {
  private model: string;
  private provider: 'claude' | 'openai' | 'gemini';

  async generate(prompt: string, options?: GenerateOptions): Promise<string> {
    // 根据配置选择 provider
    switch (this.provider) {
      case 'claude':
        return this.generateWithClaude(prompt, options);
      case 'openai':
        return this.generateWithOpenAI(prompt, options);
      case 'gemini':
        return this.generateWithGemini(prompt, options);
    }
  }

  // 流式响应
  async *generateStream(prompt: string, options?: GenerateOptions) {
    const response = await this.callProvider(prompt, options);
    for await (const chunk of response) {
      yield chunk;
    }
  }
}
```

## 四、视频生成服务（AIVideo）

### 4.1 服务封装

```typescript
// src/utils/ai/video/index.ts
class AIVideoService {
  private provider: 'seedance' | 'runway' | 'kling';

  async generate(prompt: string, imageUrl: string, options?: VideoOptions): Promise<VideoResult> {
    // 提交任务
    const taskId = await this.submitTask(prompt, imageUrl, options);

    // 轮询结果
    return this.pollResult(taskId);
  }

  private async submitTask(prompt: string, imageUrl: string, options?: VideoOptions): Promise<string> {
    switch (this.provider) {
      case 'seedance':
        return this.submitToSeedance(prompt, imageUrl, options);
      case 'runway':
        return this.submitToRunway(prompt, imageUrl, options);
      case 'kling':
        return this.submitToKling(prompt, imageUrl, options);
    }
  }
}
```

## 五、多Key轮询系统

### 5.1 负载均衡

```typescript
class MultiKeyManager {
  private keys: APIKey[];
  private currentIndex = 0;

  getNextKey(): APIKey {
    const key = this.keys[this.currentIndex];
    this.currentIndex = (this.currentIndex + 1) % this.keys.length;
    return key;
  }

  async callWithFallback(prompt: string, providers: string[]): Promise<string> {
    for (const provider of providers) {
      try {
        return await this.call(provider, prompt);
      } catch (error) {
        console.error(`${provider} failed, trying next...`);
        continue;
      }
    }
    throw new Error('All providers failed');
  }
}
```

### 5.2 自动重试

```typescript
async function withRetry<T>(
  fn: () => Promise<T>,
  options = { maxRetries: 3, delay: 1000 }
): Promise<T> {
  for (let i = 0; i < options.maxRetries; i++) {
    try {
      return await fn();
    } catch (error) {
      if (i === options.maxRetries - 1) throw error;
      await delay(options.delay * Math.pow(2, i));  // 指数退避
    }
  }
}
```

## 六、任务状态追踪

### 6.1 任务状态

```typescript
interface Task {
  id: string;
  type: 'script' | 'character' | 'shot' | 'video';
  status: 'pending' | 'processing' | 'completed' | 'failed';
  progress: number;      // 0-100
  result?: string;       // 生成结果URL
  error?: string;        // 错误信息
  startedAt?: Date;
  completedAt?: Date;
}
```

### 6.2 WebSocket推送

```typescript
// 实时推送任务进度
function emitProgress(ws: WebSocket, task: Task) {
  ws.send(JSON.stringify({
    type: 'task_progress',
    data: {
      taskId: task.id,
      status: task.status,
      progress: task.progress,
      result: task.result
    }
  }));
}
```
