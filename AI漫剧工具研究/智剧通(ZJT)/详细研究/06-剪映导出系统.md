位于 `jianying/src/core.py` 的 `JianyingMultiTrackLibrary` 是剪映草稿生成的核心库类。

```python
class JianyingMultiTrackLibrary:

                 material_path_prefix: Optional[str] = None,
                 width: int = 1920, height: int = 1080, fps: int = 30):

        # 时间轴
        self.total_duration = 0
```

## 二、核心数据结构

### 2.1 Track（轨道）

```python
@dataclass
class Track:
    track_id: str          # 唯一ID
    track_type: str        # "video" 或 "audio"
    name: str              # 轨道名称
    segments: List[MediaSegment] = field(default_factory=list)
    is_default_name: bool  # 是否使用默认名称
```

### 2.2 MediaSegment（媒体片段）

```python
@dataclass
class MediaSegment:
    file_path: str     # 源文件路径
    start_time: int   # 在轨道上的起始时间（微秒）
    duration: int     # 持续时间（微秒）
    source_start: int # 源文件中的起始位置（微秒）
    volume: float     # 音量（0.0-1.0）
    speed: float     # 播放速度（1.0=正常）
    visible: bool     # 是否可见（仅视频）
```

### 3.2 添加视频片段（核心方法）

```python
def add_video_to_track(self, track_id: str, file_path: str,
                       start_time: int, duration: int = None,
                       source_start: int = 0, volume: float = 1.0,
                       speed: float = 1.0, is_placeholder: bool = False) -> str:
    # 1. 查找轨道
    track = self._find_track(track_id, "video")

    # 2. 自动获取时长（使用ffprobe）
    if duration is None:
        duration = self.get_media_duration(file_path)

    # 3. 创建素材
    material_id = self._generate_id("VIDEO_MAT_")
    self.video_materials[material_id] = {
        "id": material_id,
        "material_id": material_id,
        "material_name": os.path.basename(file_path),
        "path": self._build_material_path(os.path.basename(file_path)),
        "type": "video",
        "duration": duration,
        "height": video_height,
        "width": video_width,
        "crop": {
            "lower_left_x": 0.0, "lower_left_y": 1.0,
            "lower_right_x": 1.0, "lower_right_y": 1.0,
            "upper_left_x": 0.0, "upper_left_y": 0.0,
            "upper_right_x": 1.0, "upper_right_y": 0.0
        }
    }

    # 4. 添加片段到轨道
    segment = MediaSegment(
        file_path=file_path,
        start_time=start_time,
        duration=duration,
        source_start=source_start,
        volume=volume if not is_placeholder else 0.0,
        speed=speed,
        visible=not is_placeholder
    )
    track.segments.append(segment)

    # 5. 更新总时长
    self.total_duration = max(self.total_duration, start_time + duration)

    return material_id
```

## 四、草稿生成器（draft_generator.py）

### 4.1 生成流程

```python
def generate_draft(self, copy_media_files: bool = True,
                   media_source_dir: Optional[str] = None) -> str:
    # 1. 创建目录结构
    draft_path = os.path.join(self.library.output_dir, self.library.draft_name)
    os.makedirs(draft_path, exist_ok=True)

    resources_path = os.path.join(draft_path, "Resources")
    local_resources_path = os.path.join(resources_path, "local")
    os.makedirs(local_resources_path, exist_ok=True)

    # 2. 生成 draft_content.json
    content_path = os.path.join(draft_path, "draft_content.json")
    with open(content_path, "w", encoding="utf-8") as f:
        json.dump(self._build_draft_content(), f, ensure_ascii=False, indent=2)

    # 3. 生成 draft_meta_info.json
    meta_path = os.path.join(draft_path, "draft_meta_info.json")
    with open(meta_path, "w", encoding="utf-8") as f:
        json.dump(self._build_draft_meta_info(), f, ensure_ascii=False, indent=2)

    # 4. 复制媒体文件
    if copy_media_files:
        self._copy_media_files(local_resources_path, media_source_dir)

    return draft_path
```

## 六、轨道构建

```python
def _build_tracks(self) -> List[Dict[str, Any]]:
    tracks = []

    # 视频轨道
    for track in self.library.video_tracks:
        track_data = {
            "attribute": 0,
            "flag": 0,
            "id": track.track_id,
            "is_default_name": track.is_default_name,
            "name": track.name,
            "segments": self._build_video_segments(track),
            "type": "video"
        }
        tracks.append(track_data)

    # 音频轨道
    for track in self.library.audio_tracks:
        track_data = {
            "id": track.track_id,
            "name": track.name,
            "segments": self._build_audio_segments(track),
            "type": "audio"
        }
        tracks.append(track_data)

    return tracks
```

## 七、视频片段构建

```python
def _build_video_segments(self, track) -> List[Dict[str, Any]]:
    segments = []

    for segment in track.segments:
        material_id = self._find_material_id(segment.file_path, "video")
        if not material_id:
            continue

        segment_data = {
            "cartoon": False,
            "clip": {
                "alpha": 1.0,
                "flip": {"horizontal": False, "vertical": False},
                "rotation": 0.0,
                "scale": {"x": 1.0, "y": 1.0},
                "transform": {"x": 0.0, "y": 0.0}
            },

            "id": self.library._generate_id("SEGMENT_"),
            "material_id": material_id,  # 关联素材

            # 源时间范围
            "source_timerange": {
                "duration": segment.duration,
                "start": segment.source_start
            },

            "speed": segment.speed,

            # 目标时间范围
            "target_timerange": {
                "duration": int(segment.duration / segment.speed),
                "start": segment.start_time
            },

            "visible": segment.visible,
            "volume": segment.volume
        }
        segments.append(segment_data)

    return segments
```

## 九、媒体时长获取

使用 ffprobe 获取准确的媒体文件时长：

```python
def get_media_duration(self, file_path: str) -> int:
    """获取媒体文件时长（微秒）"""
    cmd = [
        self.config.ffprobe_path,
        '-v', 'quiet',
        '-show_entries', 'format=duration',
        '-of', 'default=noprint_wrappers=1:nokey=1',
        file_path
    ]

    result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)

    if result.returncode == 0 and result.stdout.strip():
        duration_seconds = float(result.stdout.strip())
        return int(duration_seconds * 1000000)  # 转换为微秒

    raise RuntimeError(f"无法获取媒体文件时长: {file_path}")
```

## 十一、与漫舟 manzhou-export.md 的对比

| 维度 | ZJT | 漫舟 |
|------|-----|------|
| 导出格式 | 剪映草稿 JSON | 剪映草稿 XML/JSON |
| 素材引用 | UUID + 相对路径 | URL |
| 时间单位 | 微秒 | 秒 |
| 多轨道 | 视频轨+音频轨 | 单一轨 |
| 云上传 | 支持 | 不支持 |
| 时长获取 | ffprobe 自动检测 | 固定值 |
| 媒体复制 | 自动复制到 Resources/local | 不复制 |

**关键发现**：ZJT 的草稿格式使用 UUID 作为素材 ID，通过 `material_id` 关联素材和片段。漫舟需要实现类似的关联机制。
