## 二、画布渲染引擎（canvas.js）

### 2.1 变换控制

```javascript
class CanvasRenderer {
    constructor(canvasId) {
        this.canvas = document.getElementById(canvasId);
        this.ctx = this.canvas.getContext('2d');

        // 画布变换状态
        this.panX = 0;        // 平移X
        this.panY = 0;        // 平移Y
        this.scale = 1;       // 缩放比例
        this.minScale = 0.1;
        this.maxScale = 3;
    }

    // 世界坐标转屏幕坐标
    worldToScreen(x, y) {
        return {
            x: x * this.scale + this.panX,
            y: y * this.scale + this.panY
        };
    }

    // 屏幕坐标转世界坐标
    screenToWorld(x, y) {
        return {
            x: (x - this.panX) / this.scale,
            y: (y - this.panY) / this.scale
        };
    }
}
```

### 2.2 小地图

```javascript
renderMinimap() {
    const ctx = minimap.getContext('2d');
    ctx.fillStyle = '#1a1a1a';
    ctx.fillRect(0, 0, minimap.width, minimap.height);

    // 计算缩放比例
    const scaleX = minimap.width / this.totalWidth;
    const scale = Math.min(scaleX, scaleY);

    // 绘制节点
    for (const node of this.nodes) {
        ctx.fillStyle = this.getNodeColor(node.type);
        ctx.fillRect(node.x * scale, node.y * scale,
                     node.width * scale, node.height * scale);
    }

    // 绘制视口框
    ctx.strokeStyle = '#fff';
    ctx.strokeRect(viewportX, viewportY, viewportW, viewportH);
}
```

## 四、分镜图生成与角色匹配（shot_frame_generator.js）

这是 ZJT 前端最核心的模块，负责将分镜描述转换为 AI 生图请求。

### 4.1 角色匹配核心逻辑

```javascript
async function generateShotFrameImage(nodeId, node) {
    const imagePrompt = node.imagePrompt;

    // 1. 解析【【角色名】】标记
    const characterPattern = /【【([^】]+)】】/g;
    const characterNames = [];
    while ((match = characterPattern.exec(imagePrompt)) !== null) {
        characterNames.push(match[1].trim());
    }

    // 2. 查询角色API获取参考图
    for (const characterName of characterNames) {
        const response = await fetch(
            `/api/characters?world_id=${worldId}&keyword=${encodeURIComponent(characterName)}`
        );
        const data = await response.json();
        if (data.characters && data.characters.length > 0) {
            const character = data.characters[0];
            characterRefs.push({
                name: characterName,
                referenceImage: character.reference_image
            });
        }
    }

    // 3. 收集场景和道具参考图
    const sceneRefs = collectSceneReferences(node);
    const propsRefs = collectPropsReferences(node);

    // 4. 构建完整提示词
    const fullPrompt = buildShotFramePrompt(imagePrompt, characterRefs, sceneRefs, propsRefs);

    // 5. 调用生成API
    return await callImageGenerationAPI(fullPrompt, characterRefs);
}
```

### 4.2 参考图收集

```javascript
function collectSceneReferences(node) {
    const refs = [];
    const connectedNodes = getConnectedNodes(node.id);

    for (const connNode of connectedNodes) {
        if (connNode.type === NODE_TYPES.LOCATION) {
            if (connNode.referenceImage) {
                refs.push({
                    name: connNode.name,
                    imageUrl: connNode.referenceImage,
                    type: 'scene'
                });
            }
        }
    }
    return refs;
}

function collectPropsReferences(node) {
    const refs = [];
    const connectedNodes = getConnectedNodes(node.id);

    for (const connNode of connectedNodes) {
        if (connNode.type === NODE_TYPES.PROPS) {
            if (connNode.referenceImage) {
                refs.push({
                    name: connNode.name,
                    imageUrl: connNode.referenceImage,
                    type: 'props'
                });
            }
        }
    }
    return refs;
}
```

### 4.3 提示词构建

```javascript
function buildShotFramePrompt(basePrompt, characterRefs, sceneRefs, propsRefs) {
    let fullPrompt = basePrompt;

    // 添加角色参考说明
    if (characterRefs.length > 0) {
        const characterSection = characterRefs.map(ref =>
            `Character Reference: [Character: ${ref.name}]\nImage: ${ref.imageUrl}`
        ).join('\n');
        fullPrompt += '\n\n---\nCHARACTER REFERENCES:\n' + characterSection;
    }

    // 添加场景参考说明
    if (sceneRefs.length > 0) {
        fullPrompt += '\n\nSCENE REFERENCES:\n' +
            sceneRefs.map(r => `Scene Reference: ${r.imageUrl}`).join('\n');
    }

    // 添加道具参考说明
    if (propsRefs.length > 0) {
        fullPrompt += '\n\nPROPS REFERENCES:\n' +
            propsRefs.map(r => `Props Reference: ${r.name} - ${r.imageUrl}`).join('\n');
    }

    return fullPrompt;
}
```

### 4.4 相机描述生成

```javascript
function convertCameraToPrompt(camera) {
    const parts = [];

    // Yaw 转换（0度为正面）
    if (camera.yaw <= -75) parts.push('正侧面轮廓视图，人物面朝右');
    else if (camera.yaw > -45 && camera.yaw <= -5) parts.push('四分之三侧面视图，人物面朝右');
    else if (camera.yaw > -5 && camera.yaw < 5) parts.push('正面视图');
    else if (camera.yaw >= 5 && camera.yaw < 45) parts.push('四分之三侧面视图，人物面朝左');

    // Dolly 转换 - 景别
    if (camera.dolly >= 0 && camera.dolly < 2) parts.push('极远景');
    else if (camera.dolly >= 4 && camera.dolly < 6) parts.push('中景');
    else if (camera.dolly >= 6 && camera.dolly < 8) parts.push('近景');
    else if (camera.dolly >= 8) parts.push('特写');

    // Pitch 转换 - 拍摄角度
    if (camera.pitch <= -45) parts.push('极低角度仰视');
    else if (camera.pitch > -10 && camera.pitch < 10) parts.push('水平视线');
    else if (camera.pitch >= 45) parts.push('极高角度俯视，上帝视角');

    return parts.join('，');
}
```

### 5.2 状态轮询

```javascript
async function pollWorkflowNodeStatus(nodeId) {
    while (node.status === 'pending' || node.status === 'running') {
        const response = await fetch(`/api/workflow/nodes/${nodeId}/status`);
        const status = await response.json();

        node.status = status.state;
        if (status.state === 'success') {
            node.thumbnail = status.thumbnail;
            node.videoUrl = status.video_url;
            break;
        }

        await sleep(3000);
    }
}
```

## 七、与漫舟前端对比

| 维度 | ZJT | 漫舟（ai-drama-studio） |
|------|-----|------|
| 渲染技术 | Canvas + SVG 双层 | React + Canvas |
| 节点类型 | 13种（含特效/滤镜） | 基础节点 |
| 角色匹配 | 自动解析【【】】+ API查询 | Prompt 拼接 |
| 场景/道具参考 | 节点连接自动收集 | 手动拼接 |
| 工作流保存 | JSON 序列化 | JSON 序列化 |
| 小地图 | 有 | 有 |
| 相机控制 | 3D 预览 | 无 |

**关键发现**：ZJT 的【【】】标记系统在前后端都有完整支持，前端解析标记→API查询参考图→构建完整提示词，这是漫舟需要学习的核心设计。
