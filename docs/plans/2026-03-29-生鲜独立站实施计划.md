**Step 1: 编写 strapi.ts（统一的 Strapi API 客户端）**

```typescript
// apps/web/src/lib/strapi.ts
// 规范参考：docs/plans/2026-03-29-生鲜独立站技术规范.md §2.1

const STRAPI_URL = process.env.NEXT_PUBLIC_STRAPI_URL ?? 'http://localhost:1337'
const STRAPI_TOKEN = process.env.STRAPI_API_TOKEN ?? ''

interface StrapiQueryParams {
  populate?: string | string[] | Record<string, unknown>
  filters?: Record<string, unknown>
  sort?: string | string[]
  pagination?: { page?: number; pageSize?: number }
  fields?: string[]
  locale?: string
}

function buildQueryParams(params: StrapiQueryParams): URLSearchParams {
  const searchParams = new URLSearchParams()

  if (params.populate) {
    if (typeof params.populate === 'string') {
      searchParams.set('populate', params.populate)
    } else if (Array.isArray(params.populate)) {
      searchParams.set('populate', params.populate.join(','))
    } else {
      searchParams.set('populate', JSON.stringify(params.populate))
    }
  }
  if (params.filters) {
    searchParams.set('filters', JSON.stringify(params.filters))
  }
  if (params.sort) {
    searchParams.set('sort', Array.isArray(params.sort) ? params.sort.join(',') : params.sort)
  }
  if (params.pagination) {
    if (params.pagination.page) searchParams.set('pagination[page]', String(params.pagination.page))
    if (params.pagination.pageSize) searchParams.set('pagination[pageSize]', String(params.pagination.pageSize))
  }
  if (params.fields) {
    searchParams.set('fields', params.fields.join(','))
  }
  if (params.locale) {
    searchParams.set('locale', params.locale)
  }

  return searchParams
}

export async function fetchStrapi<T>(
  endpoint: string,
  params: StrapiQueryParams = {},
  options: RequestInit = {}
): Promise<T> {
  const queryString = buildQueryParams(params).toString()
  const url = `${STRAPI_URL}/api/${endpoint}${queryString ? `?${queryString}` : ''}`

  const headers: HeadersInit = {
    'Content-Type': 'application/json',
    ...(STRAPI_TOKEN && { Authorization: `Bearer ${STRAPI_TOKEN}` }),
    ...options.headers,
  }

  const response = await fetch(url, {
    ...options,
    headers,
    next: { revalidate: 86400 }, // 默认 ISR 24h
  } as RequestInit)

  if (!response.ok) {
    throw new Error(`Strapi API Error: ${response.status} ${response.statusText} for ${url}`)
  }

  const json = await response.json()
  return json.data as T
}

// ─── CityPage 查询 ───
export async function getCityPage(slug: string) {
  return fetchStrapi<CityPage>('city-pages', {
    filters: { slug: { $eq: slug }, isActive: { $eq: true } },
    populate: {
      heroImage: '*',
      galleryImages: '*',
      seoMetadata: '*',
      manager: { populate: { weComQrCode: '*' } },
      userReviews: { populate: '*' },
    },
  })
}

export async function getAllCitySlugs(): Promise<string[]> {
  const cities = await fetchStrapi<Array<{ slug: string }>>('city-pages', {
    filters: { isActive: { $eq: true } },
    fields: ['slug'],
    pagination: { pageSize: 100 },
  })
  return cities.map((c) => c.slug)
}

// ─── Article 查询 ───
export async function getArticles(params: { page?: number; pageSize?: number; category?: string } = {}) {
  const filters: Record<string, unknown> = {}
  if (params.category) filters.category = { $eq: params.category }

  return fetchStrapi<Article[]>('articles', {
    filters,
    sort: [{ publishedAt: 'desc' }],
    populate: { coverImage: '*', seoMetadata: '*' },
    pagination: { page: params.page ?? 1, pageSize: params.pageSize ?? 10 },
  })
}

export async function getArticle(slug: string) {
  return fetchStrapi<Article>('articles', {
    filters: { slug: { $eq: slug } },
    populate: { coverImage: '*', seoMetadata: '*', relatedProducts: { populate: { images: '*' } } },
  })
}

// ─── Product 查询 ───
export async function getProducts() {
```

```tsx
// apps/web/src/components/ui/WeComQRWidget.tsx
'use client'

import { useState, useEffect } from 'react'
import { motion, AnimatePresence } from 'framer-motion'
import Image from 'next/image'
import { trackEvent } from '@/lib/analytics'

interface CityManager {
  name: string
  weComQrCode: { url: string }
  intro?: string
}

interface Props {
  defaultManager: CityManager
  cityManagers?: Record<string, CityManager>
  citySlug?: string
}

export default function WeComQRWidget({
  defaultManager,
  cityManagers = {},
  citySlug,
}: Props) {
  const [isOpen, setIsOpen] = useState(false)
  const [activeManager, setActiveManager] = useState<CityManager>(defaultManager)

  useEffect(() => {
    // 优先级1：URL 参数
    const urlParams = new URLSearchParams(window.location.search)
    const paramCity = urlParams.get('manager')
    if (paramCity && cityManagers[paramCity]) {
      setActiveManager(cityManagers[paramCity])
      trackEvent('wecom_qr_shown', { citySlug: paramCity, managerName: cityManagers[paramCity].name })
      return
    }

    // 优先级2：当前城市 slug
    if (citySlug && cityManagers[citySlug]) {
      setActiveManager(cityManagers[citySlug])
    }

    trackEvent('wecom_qr_shown', { citySlug: citySlug ?? 'default', managerName: activeManager.name })
  }, [citySlug, cityManagers, defaultManager])

  const handleClick = () => {
    setIsOpen(!isOpen)
    if (!isOpen) {
      trackEvent('wecom_qr_clicked', { citySlug: citySlug ?? 'default', managerName: activeManager.name })
    }
  }

  return (
    <div className="fixed bottom-6 right-6 z-50 flex flex-col items-end gap-3">
      {/* 展开面板 */}
      <AnimatePresence>
        {isOpen && (
          <motion.div
            initial={{ opacity: 0, scale: 0.9, y: 10 }}
            animate={{ opacity: 1, scale: 1, y: 0 }}
            exit={{ opacity: 0, scale: 0.9, y: 10 }}
            transition={{ duration: 0.25, ease: 'easeOut', damping: 25 }}
            className="bg-white rounded-lg shadow-xl p-4 w-64 text-center"
```

```typescript
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY })

const ARTICLE_PROMPT_TEMPLATE = `【角色】
你是品牌"品牌名"（高端健康生鲜）的资深内容编辑，精通 SEO 写作和用户心理。

3. 结尾 CTA：引导添加「{{city_name}}专属管家」（语气：克制+高级感）

【SEO 要求】

- Meta Description：{{city_name}}精英家庭的{{hero_product}}选购建议，附{{city_name}}专属配送服务介绍
- 正文中 2% 关键词密度

【字数】
{{word_count}}-{{word_count + 200}} 字`

interface ArticleTopic {
  title: string
  summary: string
  keywords: string[]
  category: '科普' | '食谱' | '动态'
}

interface GeneratedArticle {
  title: string
  metaTitle: string
  metaDescription: string
  summary: string
  content: string
  keywords: string[]
  category: '科普' | '食谱' | '动态'
  cityScope: string
  imagePrompt: string
}

const CITIES = [
  '上海', '北京', '广州', '深圳',
  '成都', '杭州', '重庆', '武汉', '西安', '苏州',
]

const CATEGORIES: Array<'科普' | '食谱' | '动态'> = ['科普', '食谱', '动态']

export async function generateTopic(city: string, category: string): Promise<ArticleTopic> {
  const prompt = `请为品牌"品牌名"生成一个${category}类文章选题：
城市：${city}
品类：${category === '科普' ? '牛油果/浆果等健康生鲜' : category === '食谱' ? '健康早餐/减脂餐' : '品牌动态/产地故事'}
要求：1句话标题，2-3句摘要，3-5个SEO关键词，用JSON格式输出`

  const response = await openai.chat.completions.create({
    model: 'gpt-4o',
    messages: [{ role: 'user', content: prompt }],
    response_format: { type: 'json_object' },
    temperature: 0.8,
  })

  return JSON.parse(response.choices[0].message.content ?? '{}') as ArticleTopic
}

export async function generateArticle(topic: ArticleTopic, city: string): Promise<GeneratedArticle> {
  const prompt = ARTICLE_PROMPT_TEMPLATE
    .replace('{{city_name}}', city)
    .replace('{{hero_product}}', '即食牛油果')
    .replace('{{seo_keywords}}', topic.keywords.join('、'))
    .replace('{{category}}', topic.category)
    .replace('{{word_count}}', '800')

  const response = await openai.chat.completions.create({
    model: 'gpt-4o',
    messages: [{ role: 'user', content: prompt }],
    response_format: { type: 'json_object' },
    temperature: 0.3, // 较低温度保证质量稳定
  })

  const result = JSON.parse(response.choices[0].message.content ?? '{}')

  return {
    title: result.title,
    metaTitle: result.metaTitle ?? `${city}高端健康生鲜指南 | 品牌名`,
    metaDescription: result.metaDescription ?? '',
    summary: result.summary ?? '',
    content: result.content ?? '',
    keywords: topic.keywords,
    category: topic.category,
    cityScope: city,
    imagePrompt: `极简风格，${result.title}相关内容，置于白色背景，自然光，微距摄影，高端食品杂志排版感，无文字，无水印 --ar 16:9 --v 6.1`,
  }
}

export async function runPipeline() {
  const articles: GeneratedArticle[] = []

  for (const city of CITIES) {
    for (const category of CATEGORIES) {
      console.log(`[${city}] 生成${category}文章...`)
      const topic = await generateTopic(city, category)
      const article = await generateArticle(topic, city)
      articles.push(article)
      console.log(`[${city}] ✓ ${article.title}`)
    }
  }

  return articles
}
```

```typescript
// scripts/ai-pipeline/src/upload-to-strapi.ts
import type { GeneratedArticle } from './generate-articles'

const STRAPI_URL = process.env.STRAPI_API_URL ?? 'http://localhost:1337'
const STRAPI_TOKEN = process.env.STRAPI_API_TOKEN ?? ''

interface UploadResult {
  success: boolean
  articleId?: number
  error?: string
}

export async function uploadArticle(article: GeneratedArticle, coverImageUrl: string): Promise<UploadResult> {
  const response = await fetch(`${STRAPI_URL}/api/articles`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${STRAPI_TOKEN}`,
    },
    body: JSON.stringify({
      data: {
        title: article.title,
        slug: article.title.replace(/[^a-zA-Z0-9\u4e00-\u9fa5]/g, '-'),
        category: article.category,
        summary: article.summary,
        metaDescription: article.metaDescription,
        content: article.content,
        coverImage: article.coverImageUrl,
        cityScope: article.cityScope,
        isAiGenerated: true,
        aiPromptUsed: `Topic: ${article.title}`,
        author: '品牌编辑部',
        readingTime: Math.ceil(article.content.length / 400),
        // publishedAt 留空 = 草稿状态
      },
    }),
  })

  if (!response.ok) {
    const errorText = await response.text()
    return { success: false, error: errorText }
  }

  const json = await response.json()
  return { success: true, articleId: json.data?.id }
}

export async function uploadAllArticles(
  articles: GeneratedArticle[],
  imageUrls: string[],
): Promise<void> {
  for (let i = 0; i < articles.length; i++) {
    const article = articles[i]
    const imageUrl = imageUrls[i]
    const result = await uploadArticle({ ...article, coverImageUrl: imageUrl }, imageUrl)
    if (result.success) {
      console.log(`✓ 上传成功: ${article.title} (ID: ${result.articleId})`)
    } else {
      console.error(`✗ 上传失败: ${article.title} — ${result.error}`)
    }
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { revalidatePath } from 'next/cache'

export async function POST(request: NextRequest) {
  const secret = request.headers.get('x-revalidation-secret')
  if (secret !== process.env.REVALIDATION_SECRET) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  try {
    const body = await request.json()
    const { type, slug } = body

    if (type === 'city-page' && slug) {
      revalidatePath(`/${slug}`)
      revalidatePath('/sitemap.xml')
    } else if (type === 'article') {
      revalidatePath('/blog')
      revalidatePath('/sitemap.xml')
    } else if (type === 'product') {
      revalidatePath('/products')
    } else {
      revalidatePath('/')
    }

    return NextResponse.json({ revalidated: true, timestamp: new Date().toISOString() })
  } catch {
    return NextResponse.json({ error: 'Failed to revalidate' }, { status: 500 })
  }
}
```
