**原理**: 将原始视频（4K/1080P/任意格式）标准化为 720p + 12fps + H.264，大幅减少处理量的同时保留足够视觉信息供 AI 分析。

- [ ] **Step 1: 创建 ffmpeg_preprocess.py**

```python
"""ai-drama-studio/backend/pipeline/ffmpeg_preprocess.py"""
import subprocess
import os
import uuid
from pathlib import Path
from config import config

class FFmpegPreprocessor:
    """FFmpeg 视频预处理管线：标准化 → 压缩 → 格式统一"""

    def __init__(self):
        self.height = config.STANDARDIZED_HEIGHT  # 720
        self.fps = config.STANDARDIZED_FPS          # 12

    def standardize(self, input_path: str) -> tuple[str, dict]:
        """
        执行 FFmpeg 标准化转换。

        Returns:
            output_path: 标准化后的视频路径
            metadata: 视频元数据（时长、分辨率、帧率）
        """
        video_id = uuid.uuid4().hex[:8]
        output_filename = f"std_{video_id}.mp4"
        output_path = os.path.join(config.STANDARDIZED_DIR, output_filename)

        # 获取原始元数据
        metadata = self._get_metadata(input_path)

        # FFmpeg 命令
        cmd = [
            "ffmpeg",
            "-y",                              # 覆盖输出
            "-i", input_path,                  # 输入文件
            "-vf", f"scale=-2:{self.height},fps={self.fps}",  # 等比缩放至720p，12fps
            "-c:v", "libx264",
            "-preset", "veryfast",
            "-crf", "28",                     # 压缩质量（文件大小 vs 质量）
            "-c:a", "aac",
            "-b:a", "128k",
            "-progress", "pipe:1",             # 输出进度到 stdout
            output_path
        ]

        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True
        )

        if result.returncode != 0:
            raise RuntimeError(f"FFmpeg failed: {result.stderr}")

        # 获取输出文件元数据
        output_metadata = self._get_metadata(output_path)
        output_metadata["output_path"] = output_path

        return output_path, output_metadata

    def _get_metadata(self, video_path: str) -> dict:
        """用 ffprobe 获取视频元数据"""
        cmd = [
            "ffprobe",
            "-v", "quiet",
            "-print_format", "json",
            "-show_format",
            "-show_streams",
            video_path
        ]
        import json, subprocess
        result = subprocess.run(cmd, capture_output=True, text=True)
        data = json.loads(result.stdout)

        video_stream = next(
            (s for s in data["streams"] if s["codec_type"] == "video"), {}
        )
        duration = float(data["format"].get("duration", 0))

        return {
            "duration": duration,
            "width": video_stream.get("width", 0),
            "height": video_stream.get("height", 0),
            "fps": eval(video_stream.get("r_frame_rate", "0/1")),  # "30/1" → 30.0
        }
```

"""ai-drama-studio/backend/pipeline/scene_detect.py"""
from scenedetect import open_video, SceneManager, ContentDetector
from scenedetect.detectors import ContentDetector
from config import config
import logging

logger = logging.getLogger("pyscenedetect")

class SceneDetector:
    """PySceneDetect 镜头边界检测器

    使用 ContentDetector，基于 HSV 色彩空间逐帧比对，
    检测画面主体/场景变化触发镜头切分。
    """

    def __init__(self):
        self.threshold = config.SCENE_THRESHOLD  # 27.0（短视频/动画推荐值）
        self.min_scene_len = config.MIN_SCENE_LEN  # 15帧，防止误检

    def detect(self, video_path: str) -> list[dict]:
        """
        执行镜头边界检测。

        Returns:
            List[dict]: 每个镜头的 {shot_id, start_time, end_time, duration_sec}
        """
        video = open_video(video_path)
        scene_manager = SceneManager()

        detector = ContentDetector(
            threshold=self.threshold,
            min_scene_len=self.min_scene_len,
        )
        scene_manager.add_detector(detector)
        scene_manager.detect_scenes(video, show_progress=False)

        scene_list = scene_manager.get_scene_list()

        shots = []
        for i, (start, end) in enumerate(scene_list):
            start_sec = start.get_seconds()
            end_sec = end.get_seconds()
            duration = end_sec - start_sec

            shots.append({
                "shot_id": i + 1,
                "start_time": start_sec,
                "end_time": end_sec,
                "duration_sec": round(duration, 3),
                "start_frame": start.get_frames(),
                "end_frame": end.get_frames(),
            })

        logger.info(f"检测到 {len(shots)} 个镜头")
        return shots
```

```python
"""ai-drama-studio/backend/pipeline/frame_extract.py"""
import cv2
import os
import uuid
from config import config

class FrameExtractor:
    """动态抽帧器：根据镜头时长智能选择抽帧位置"""

    def extract_shot_frames(
        self,
        video_path: str,
        fps: float,
        shots: list[dict],
        output_dir: str,
    ) -> list[dict]:
        """
        对每个镜头执行动态抽帧。

        抽帧策略（与方案文档完全一致）：
        - 时长 < 1s:   抽 1 帧（中间帧）
        - 1s ≤ 时长 ≤ 3s: 抽 2 帧（前20% + 后80%）
        - 时长 > 3s:  抽 3 帧（前10% + 中间 + 后90%）

        Returns:
            List[dict]: 每个镜头的 frame_paths（含相对URL）
        """
        os.makedirs(output_dir, exist_ok=True)
        cap = cv2.VideoCapture(video_path)

        if not cap.isOpened():
            raise RuntimeError(f"无法打开视频: {video_path}")

        results = []

        for shot in shots:
            start_frame = shot["start_frame"]
            end_frame = shot["end_frame"]
            duration_frames = end_frame - start_frame
            duration_sec = shot["duration_sec"]

            target_positions = self._smart_positions(duration_frames, duration_sec)
            frame_paths = []

            for j, pos_ratio in enumerate(target_positions):
                frame_idx = start_frame + int(duration_frames * pos_ratio)
                cap.set(cv2.CAP_PROP_POS_FRAMES, frame_idx)
                ret, frame = cap.read()

                if ret:
                    filename = f"shot_{shot['shot_id']:03d}_f{j+1}.jpg"
                    img_path = os.path.join(output_dir, filename)
                    # 质量 95%，避免 JPEG 伪影影响 AI 分析
                    cv2.imwrite(img_path, frame, [cv2.IMWRITE_JPEG_QUALITY, 95])
                    frame_paths.append({
                        "filename": filename,
                        "url": f"/frames/{filename}",
                        "frame_index": frame_idx,
                        "position_ratio": pos_ratio,
                    })

            results.append({
                "shot_id": shot["shot_id"],
                "frame_count": len(frame_paths),
                "frames": frame_paths,
            })

        cap.release()
        return results

    def _smart_positions(self, duration_frames: int, duration_sec: float) -> list[float]:
        """根据镜头时长返回抽帧位置比例列表"""
        if duration_sec < 1.0:
            return [0.5]
        elif duration_sec <= 3.0:
            return [0.2, 0.8]
        else:
            return [0.1, 0.5, 0.9]
```

```python
"""ai-drama-studio/backend/pipeline/ai_analyzer.py"""
import json
import httpx
import anthropic
import google.generativeai as genai
from config import config
from prompts.shot_analysis import SHOT_ANALYSIS_SYSTEM_PROMPT, build_shot_analysis_user_prompt
import asyncio
import logging

logger = logging.getLogger(__name__)

class AIAnalyzer:
    """
    AI 分镜分析器，支持 Claude 和 Gemini 双后端。
    提供 SSE 流式输出，前端可实时渲染 JSON 字段。
    """

    def __init__(self):
        self.model_type = config.AI_MODEL  # "claude" | "gemini"
        self._setup_clients()

    def _setup_clients(self):
        if self.model_type == "claude":
            self.client = anthropic.Anthropic(api_key=config.ANTHROPIC_API_KEY)
        elif self.model_type == "gemini":
            genai.configure(api_key=config.GOOGLE_API_KEY)
            self.client = genai.GenerativeModel(config.GEMINI_MODEL)

    async def analyze_shot_streaming(
        self,
        shot_id: int,
        start_time: float,
        end_time: float,
        duration: float,
        frame_urls: list[str],
        shot_context: str = "",
    ) -> AsyncGenerator[str, None]:
        """
        流式分析单个镜头，yield SSE 格式的数据块。

        SSE 事件类型：
        - shot_start: 镜头开始分析
        - chunk: 文本块（用于打字机效果）
        - shot_complete: 完整 JSON 结果
        - error: 错误信息
        """
        user_prompt = build_shot_analysis_user_prompt(
            shot_id, start_time, end_time, duration, frame_urls, shot_context
        )

        if self.model_type == "claude":
            async for event in self._claude_stream(shot_id, user_prompt):
                yield event
        elif self.model_type == "gemini":
            async for event in self._gemini_stream(shot_id, user_prompt):
                yield event

    async def _claude_stream(self, shot_id: int, user_prompt: str):
        """Claude SSE 流式"""
        system = SHOT_ANALYSIS_SYSTEM_PROMPT

        # 注入 shot_id 到 System Prompt
        system_with_id = system.replace(
            '"shot_id": "[自动填入，由请求参数传入]"',
            f'"shot_id": "{shot_id}"'
        )

        try:
            async with self.client.messages.stream(
                model=config.CLAUDE_MODEL,
                system=system_with_id,
                max_tokens=4096,
                messages=[{"role": "user", "content": user_prompt}],
            ) as stream:
                accumulated = ""
                async with httpx.AsyncClient() as http_client:
                    async for text in stream.text_stream:
                        accumulated += text
                        # SSE 格式：data: {...}\n\n
                        yield f'data: {json.dumps({"type": "chunk", "shot_id": shot_id, "text": text}, ensure_ascii=False)}\n\n'
        except Exception as e:
            logger.error(f"Claude API error: {e}")
            yield f'data: {json.dumps({"type": "error", "shot_id": shot_id, "message": str(e)}, ensure_ascii=False)}\n\n'

    async def _gemini_stream(self, shot_id: int, user_prompt: str):
        """Gemini SSE 流式"""
        try:
            response = self.client.generate_content(
                user_prompt,
                generation_config={
                    "system_instruction": SHOT_ANALYSIS_SYSTEM_PROMPT,
                    "response_mime_type": "text/plain",
                },
                stream=True,
            )
            for chunk in response:
                yield f'data: {json.dumps({"type": "chunk", "shot_id": shot_id, "text": chunk.text}, ensure_ascii=False)}\n\n'
        except Exception as e:
            logger.error(f"Gemini API error: {e}")
            yield f'data: {json.dumps({"type": "error", "shot_id": shot_id, "message": str(e)}, ensure_ascii=False)}\n\n'

    async def analyze_shot_sync(self, shot_id: int, start_time: float, end_time: float,
                                duration: float, frame_urls: list[str],
                                shot_context: str = "") -> dict:
        """同步分析单个镜头，返回完整 JSON"""
        user_prompt = build_shot_analysis_user_prompt(
            shot_id, start_time, end_time, duration, frame_urls, shot_context
        )

        system = SHOT_ANALYSIS_SYSTEM_PROMPT.replace(
            '"shot_id": "[自动填入，由请求参数传入]"',
            f'"shot_id": "{shot_id}"'
        )

        if self.model_type == "claude":
            response = self.client.messages.create(
                model=config.CLAUDE_MODEL,
                system=system,
                max_tokens=4096,
                messages=[{"role": "user", "content": user_prompt}],
            )
            text = response.content[0].text.strip()
            # 去掉可能的 ```json 包裹
            if text.startswith("```"):
                text = text.split("\n", 1)[1]
            if text.endswith("```"):
                text = text[:-3]
            return json.loads(text)

        elif self.model_type == "gemini":
            response = self.client.generate_content(
                user_prompt,
                generation_config={
                    "system_instruction": SHOT_ANALYSIS_SYSTEM_PROMPT,
                },
            )
            text = response.text.strip()
            if text.startswith("```"):
                text = text.split("\n", 1)[1]
            if text.endswith("```"):
                text = text[:-3]
            return json.loads(text)

async def run_pipeline(job_id: str, input_path: str, shot_context: str):

        pipeline_state["phase"] = "preprocessing"

        shots = detector.detect(std_path)
        pipeline_state["total_shots"] = len(shots)
        pipeline_state["shots"] = shots
        pipeline_state["progress"] = 40
        _save_job_state(job_id, pipeline_state)

        # ── Phase 3: 动态抽帧 ──
        pipeline_state["phase"] = "frame_extraction"
        pipeline_state["phase_desc"] = f"提取关键帧..."
        _save_job_state(job_id, pipeline_state)

        extractor = FrameExtractor()
        frame_results = extractor.extract_shot_frames(
            video_path=std_path,
            fps=metadata["fps"],
            shots=shots,
            output_dir=frame_dir,
        )
        pipeline_state["frame_results"] = frame_results
        pipeline_state["progress"] = 60
        _save_job_state(job_id, pipeline_state)

        # ── Phase 4: AI 分镜分析 ──
        pipeline_state["phase"] = "ai_analysis"
        pipeline_state["phase_desc"] = "AI分镜分析中..."

        analyzed_shots = []

                end_time=shot["end_time"],
                duration=shot["duration_sec"],

                shot_context=shot_context,
            )

            result["extracted_frames"] = frame_result["frames"]
            analyzed_shots.append(result)

            # 更新进度（60% ~ 95%）
            pipeline_state["current_shot"] = shot["shot_id"]
            pipeline_state["progress"] = 60 + int(35 * (i + 1) / len(shots))
            _save_job_state(job_id, pipeline_state)

        pipeline_state["shots"] = analyzed_shots
        pipeline_state["status"] = "completed"
        pipeline_state["progress"] = 100
        pipeline_state["phase_desc"] = "分析完成！"
        _save_job_state(job_id, pipeline_state)

    except Exception as e:

        pipeline_state["status"] = "error"
        pipeline_state["error"] = str(e)

@router.get("/stream/{job_id}")
async def stream_lapian(job_id: str):
    """
    SSE 流式推送每个镜头的分析结果。
    前端可实时渲染每个镜头的 JSON 字段（打字机效果）。
    """
    async def event_generator():
        while True:
            state = _load_job_state(job_id)

            if state is None:
                yield f"data: {json.dumps({'type': 'error', 'message': 'Job不存在'})}\n\n"
                break

            if state["status"] == "completed":
                for shot in state["shots"]:
                    yield f"data: {json.dumps({'type': 'shot_complete', 'shot': shot})}\n\n"
                yield f"data: {json.dumps({'type': 'done'})}\n\n"
                break

            if state["status"] == "error":
                yield f"data: {json.dumps({'type': 'error', 'message': state['error']})}\n\n"
                break

            # 定期推送状态更新（心跳）
            yield f"data: {json.dumps({'type': 'status', 'phase': state['phase'], 'progress': state['progress'], 'phase_desc': state['phase_desc']})}\n\n"

            await asyncio.sleep(2)

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )

import tempfile

def _get_job_file(job_id: str) -> str:
    return os.path.join(tempfile.gettempdir(), f"lapian_job_{job_id}.json")

def _save_job_state(job_id: str, state: dict):
    with open(_get_job_file(job_id), "w", encoding="utf-8") as f:
        json.dump(state, f, ensure_ascii=False, indent=2)

def _load_job_state(job_id: str) -> Optional[dict]:
    path = _get_job_file(job_id)
    if not os.path.exists(path):

import { LapianJob, LapianShot } from '../types'

const API_BASE = 'http://localhost:8000/api/lapian'

export async function uploadVideo(
  file: File,
  shotContext: string = '',
  onProgress?: (progress: number, phase: string) => void,
  onShotComplete?: (shot: LapianShot) => void,
  onError?: (msg: string) => void,
): Promise<string> {
  // 1. 上传视频，获取 job_id
  const formData = new FormData()
  formData.append('file', file)
  formData.append('shot_context', shotContext)

  const resp = await fetch(`${API_BASE}/analyze`, {
    method: 'POST',
    body: formData,
  })
  const { job_id } = await resp.json()

  // 2. 建立 SSE 连接，实时接收结果
  const es = new EventSource(`${API_BASE}/stream/${job_id}`)

  es.addEventListener('status', (e) => {
    const data = JSON.parse(e.data)
    onProgress?.(data.progress, data.phase_desc)
  })

  es.addEventListener('shot_complete', (e) => {
    const data = JSON.parse(e.data)
    onShotComplete?.(data.shot)
  })

  es.addEventListener('done', () => {
    es.close()
  })

  es.addEventListener('error', (e) => {
    const data = JSON.parse((e as MessageEvent).data)
    onError?.(data.message || 'SSE连接错误')
    es.close()
  })

  return job_id
}

export async function pollStatus(jobId: string): Promise<LapianJob> {
  const resp = await fetch(`${API_BASE}/status/${jobId}`)
  return resp.json()
}

import React, { useState } from 'react'
import { LapianProgress } from '../components/Lapian/LapianProgress'
import { ShotCard } from '../components/Lapian/ShotCard'
import { LapianTable } from '../components/Lapian/LapianTable'
import { LapianShot } from '../types'

type View = 'cards' | 'table'

export function LapianPage() {
  const [shots, setShots] = useState<LapianShot[]>([])
  const [uploading, setUploading] = useState(false)
  const [progress, setProgress] = useState(0)
  const [phase, setPhase] = useState('')
  const [currentShot, setCurrentShot] = useState(0)
  const [totalShots, setTotalShots] = useState(0)
  const [view, setView] = useState<View>('cards')
  const [latestShotId, setLatestShotId] = useState<number | null>(null)

  function handleShotComplete(shot: LapianShot) {
    setShots(prev => [...prev, shot])
    setLatestShotId(shot.shot_id)
    setCurrentShot(shot.shot_id)
  }

  return (
    <div style={{ maxWidth: 900, margin: '0 auto', padding: 24 }}>
      <h1 style={{ fontSize: 24, fontWeight: 700, marginBottom: 24 }}>
        🎬 一键拉片
      </h1>

      {/* 上传区 */}
      <div style={{ marginBottom: 24 }}>
        <input
          type="file"
          accept="video/*"
          disabled={uploading}
          onChange={async (e) => {
            const file = e.target.files?.[0]
            if (!file) return
            setUploading(true)
            setShots([])
            setProgress(0)

            const { uploadVideo } = await import('../api/lapian')
            await uploadVideo(
              file,
              '',
              (p, desc) => { setProgress(p); setPhase(desc) },
              (shot) => { handleShotComplete(shot) },
              (msg) => { alert(msg); setUploading(false) }
            )
            setUploading(false)
          }}
        />
      </div>

      {/* 进度条 */}
      {uploading && (
        <LapianProgress
          phase={phase}
          progress={progress}
          currentShot={currentShot}
          totalShots={totalShots}
        />
      )}

      {/* 视图切换 */}
      {shots.length > 0 && (
        <div style={{ display: 'flex', gap: 8, margin: '16px 0' }}>
          <button onClick={() => setView('cards')}>卡片视图</button>
          <button onClick={() => setView('table')}>表格视图</button>
          <span style={{ marginLeft: 'auto', fontSize: 14, color: '#666' }}>
            共 {shots.length} 个镜头
          </span>
        </div>
      )}

      {/* 渲染 */}
      {view === 'cards' ? (
        <div>
          {shots.map(shot => (
            <ShotCard
              key={shot.shot_id}
              shot={shot}
              isNew={shot.shot_id === latestShotId}
            />
          ))}
        </div>
      ) : (
        <LapianTable shots={shots} />
      )}
    </div>
  )
}
