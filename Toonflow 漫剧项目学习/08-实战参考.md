### 2. 创建 Agent 基础类

```typescript
// src/agents/myAgent.ts
import { tool } from "ai";
import { EventEmitter } from "events";
import { z } from "zod";
import u from "@/utils";

export default class MyAgent {
  readonly emitter = new EventEmitter();
  history: any[] = [];
  private projectId: number;

  constructor(projectId: number) {
    this.projectId = projectId;
  }

  // 定义工具
  myTool = tool({
    title: "myTool",
    description: "工具描述",
    inputSchema: z.object({
      param: z.string()
    }),
    execute: async ({ param }) => {
      // 工具逻辑
      const result = await u.db("t_my_table")
        .where({ projectId: this.projectId })
        .select();
      return result;
    }
  });

  // 构建工具集合
  private getAllTools() {
    return {
      myTool: this.myTool
    };
  }

  // 主入口
  async call(msg: string) {
    this.history.push({ role: "user", content: msg });

    // 构建上下文
    const context = await this.buildContext(msg);

    // 调用 AI
    const { fullStream } = await u.ai.text.stream(
      {
        system: "你的 System Prompt",
        tools: this.getAllTools(),
        messages: this.history
      },
      await u.getPromptAi("myAgent")
    );

    // 处理流式响应
    let fullResponse = "";
    for await (const item of fullStream) {
      if (item.type === "text-delta") {
        fullResponse += item.text;
        this.emitter.emit("data", item.text);
      }
    }

    this.history.push({ role: "assistant", content: fullResponse });
    return fullResponse;
  }

  private async buildContext(task: string) {
    // 获取相关数据
    const data = await u.db("t_my_table")
      .where({ projectId: this.projectId })
      .first();

    return `${task}\n\n相关数据：${JSON.stringify(data)}`;
  }
}
```

### 3. 创建 WebSocket 路由

```typescript
// src/routes/chat.ts
import express from "express";
import expressWs, { Application } from "express-ws";
import MyAgent from "@/agents/myAgent";

const router = express.Router();
expressWs(router as unknown as Application);

router.ws("/", async (ws, req) => {
  const projectId = req.query.projectId;

  if (!projectId) {
    ws.send(JSON.stringify({ type: "error", data: "项目ID缺失" }));
    ws.close();
    return;
  }

  const agent = new MyAgent(Number(projectId));

  // 事件监听
  agent.emitter.on("data", (text) => {
    ws.send(JSON.stringify({ type: "stream", data: text }));
  });

  agent.emitter.on("response", (text) => {
    ws.send(JSON.stringify({ type: "response_end", data: text }));
  });

  // 初始化完成
  ws.send(JSON.stringify({ type: "init", data: { projectId } }));

  // 消息处理
  ws.on("message", async (rawData) => {
    const data = JSON.parse(rawData);
    if (data.type === "msg") {
      await agent.call(data.data);
    }
  });
});

export default router;
```

```typescript
private safeParseJson<T>(str: string, fallback: T): T {
  try {
    return JSON.parse(str);
  } catch {
    return fallback;
  }
}

// 使用
const data = this.safeParseJson<Episode>(row.data, {});
```

### 4. 对话历史管理

```typescript
// 保存对话历史
async function saveHistory(history: any[]) {
  const existing = await u.db("t_chatHistory")
    .where({ projectId, type: "myAgent" })
    .first();

  if (existing) {
    await u.db("t_chatHistory")
      .where({ id: existing.id })
      .update({ data: JSON.stringify(history) });
  } else {
    await u.db("t_chatHistory").insert({
      projectId,
      type: "myAgent",
      data: JSON.stringify(history)
    });
  }
}
```
