# Toonflow 数据库设计

## 数据库初始化

数据库初始化在 `src/lib/initDB.ts` 中完成，使用 Knex.js 创建表结构。

## 核心表结构

### 1. t_project（项目表）

```typescript
interface t_project {
  id: number;
  name: string;           // 项目名称
  intro: string;          // 项目简介
  type: string;          // 小说类型
  artStyle: string;       // 目标短剧类型
  videoRatio: string;     // 短剧画幅
  createTime: number;
  userId: number;
}
```

### 2. t_novel（小说表）

```typescript
interface t_novel {
  id: number;
  chapterIndex: number;   // 章节序号
  reel: string;          // 分卷信息
  chapter: string;       // 章节标题
  chapterData: string;   // 章节正文内容
  projectId: number;
  createTime: number;
}
```

### 3. t_storyline（故事线表）

```typescript
interface t_storyline {
  id: number;
  name: string;
  content: string;        // 故事线内容（AI生成）
  novelIds: string;
  projectId: number;
}
```

### 4. t_outline（大纲表）

```typescript
interface t_outline {
  id: number;
  episode: number;        // 集数
  data: string;          // JSON 格式的大纲数据
  projectId: number;
}
```

**data 字段结构：**
```json
{
  "episodeIndex": 1,
  "title": "逆袭开始",
  "chapterRange": [1, 2, 3],
  "scenes": [
    { "name": "宴会厅", "description": "金碧辉煌..." }
  ],
  "characters": [
    { "name": "陈昊", "description": "旧夹克..." }
  ],
  "props": [
    { "name": "劳斯莱斯", "description": "黑色轿车..." }
  ],
  "coreConflict": "陈昊想证明自己 vs 众人嘲讽",
  "outline": "陈昊走进宴会厅...",
  "openingHook": "陈昊穿着旧夹克走进...",
  "keyEvents": ["起", "承", "转", "合"],
  "emotionalCurve": "2→5→9→3",
  "visualHighlights": ["镜头1", "镜头2"],
  "endingHook": "悬念延伸...",
  "classicQuotes": ["金句1", "金句2"]
}
```

### 5. t_script（剧本表）

```typescript
interface t_script {
  id: number;
  name: string;           // 剧本名称，如"第1集"
  content: string;         // 剧本正文内容
  projectId: number;
  outlineId: number;      // 关联的大纲ID
}
```

### 6. t_assets（资产表）

```typescript
interface t_assets {
  id: number;
  projectId: number;
  type: string;           // "角色" | "道具" | "场景"
  name: string;          // 资产名称
  intro: string;         // 资产介绍/描述
  prompt: string;        // AI 图像生成提示词
  image: string;         // 资产图片路径
  createTime: number;
}
```

### 7. t_storyboard（分镜表）

```typescript
interface t_storyboard {
  id: number;
  projectId: number;
  scriptId: number;       // 关联的剧本ID
  outlineId: number;      // 关联的大纲ID
  data: string;          // 分镜数据 JSON
  createTime: number;
}
```

### 8. t_video（视频表）

```typescript
interface t_video {
  id: number;
  projectId: number;
  resolution: string;     // 分辨率
  prompt: string;         // 视频生成提示词
  filePath: string;       // 视频文件路径
  firstFrame: string;     // 视频第一帧
  storyboardImgs: string; // 分镜图片列表
  createTime: number;
  status: number;         // 0:待生成 1:生成中 2:成功 3:失败
}
```

### 9. t_prompts（提示词配置表）

```typescript
interface t_prompts {
  id: number;
  code: string;           // 提示词代码，如 "outlineScript-main"
  name: string;          // 提示词名称
  type: string;          // "mainAgent" | "subAgent"
  parentCode: string;    // 父级提示词代码
  defaultValue: string;  // 默认提示词内容
  customValue: string;   // 自定义提示词内容（可覆盖默认）
}
```

**关键 Prompt 列表：**
| code | 类型 | 说明 |
|-----|------|------|
| outlineScript-main | mainAgent | 大纲剧本主 Agent |
| outlineScript-a1 | subAgent | 故事师 |
| outlineScript-a2 | subAgent | 大纲师 |
| outlineScript-director | subAgent | 导演 |
| storyboard-main | mainAgent | 分镜主 Agent |
| storyboard-segment | subAgent | 片段师 |
| storyboard-shot | subAgent | 分镜师 |

### 10. t_setting（设置表）

```typescript
interface t_setting {
  id: number;
  userId: number;
  tokenKey: string;       // Token 密钥
  imageModel: string;     // 图片模型配置 JSON
  languageModel: string;  // 语言模型配置 JSON
  projectId: number;
}
```

### 11. t_chatHistory（对话历史表）

```typescript
interface t_chatHistory {
  id: number;
  projectId: number;
  type: string;           // "outlineAgent" | "storyboardAgent"
  data: string;           // 对话历史 JSON
  novel: string;          // 小说章节数据 JSON
  createTime: number;
}
```

## 表关系图

```
t_project (项目)
  │
  ├── t_novel (小说章节)
  │
  ├── t_storyline (故事线)
  │
  ├── t_outline (大纲) ────────── t_script (剧本)
  │       │                         │
  │       │                         t_storyboard (分镜)
  │       │
  │       └── t_assets (资产)
  │
  ├── t_video (视频)
  │
  └── t_chatHistory (对话历史)
```

## 数据库访问方式

项目使用 Knex.js 封装数据库操作：

```typescript
// src/utils/db.ts
const db = knex({
  client: "sqlite3",
  connection: { filename: dbPath },
  useNullAsDefault: true,
});

// 使用方式
const chapters = await u.db("t_novel")
  .where({ projectId: 1 })
  .orderBy("chapterIndex", "asc")
  .select();

// 插入数据
await u.db("t_outline").insert({
  projectId: 1,
  episode: 1,
  data: JSON.stringify(outlineData)
});
```
