**积分系统**：

```typescript
// medusa/src/services/customer-points.ts
// 扩展 CustomerService，增加积分相关方法
export class CustomerPointsService {
  async addPoints(customerId: string, points: number, type: PointType, reference?: { type: string, id: string }) {
    const logRepo = this.activeManager_.getRepository('customer_points_log')
    const customer = await this.activeManager_.getRepository('customer').findOne({ where: { id: customerId } })

    const log = logRepo.create({
      id: `pt_${Date.now()}_${Math.random().toString(36).slice(2)}`,
      customer_id: customerId,
      type,
      points,
      balance_after: customer.points_balance + points,
      reference_type: reference?.type,
      reference_id: reference?.id,
      description: this.getDescription(type, points)
    })

    await logRepo.save(log)
    await this.activeManager_.getRepository('customer').update({
      where: { id: customerId },
      data: { points_balance: customer.points_balance + points }
    })

    return log
  }

  // 订单完成后增加积分（1元=1积分）
  async earnFromOrder(orderId: string) {
    const order = await this.retrieve(orderId)
    const points = Math.floor(order.total)
    await this.addPoints(order.customer_id, points, 'earn', { type: 'order', id: orderId })
    await this.checkAndUpgradeLevel(order.customer_id)
  }

  // 检查并升级会员等级
  async checkAndUpgradeLevel(customerId: string) {
    const customer = await this.retrieve(customerId)
    const levels = await this.activeManager_.getRepository('member_levels').find({ order: { min_spent: 'DESC' } })

    for (const level of levels) {
      if (customer.total_spent >= level.min_spent) {
        if (customer.member_level !== level.name) {
          await this.update(customerId, { member_level: level.name })
          // 触发升级通知
          await this.notificationService.send({
            to: customer.email,
            template: 'member-upgrade',
            data: { level: level.name, benefits: level.benefits }
          })
        }
        break
      }
    }
  }
}
```

export default async function syncProductToStrapi({ event, container }: SubscriberArgs) {
  const { id, action } = event.data
  const strapiBase = process.env.STRAPI_URL

  if (action === 'created' || action === 'updated') {
    const productModule = container.resolve(Modules.PRODUCT)
    const product = await productModule.retrieveProduct(id, {
      relations: ['images', 'variants', 'options', 'categories']
    })

    // 同步到 Strapi
    await fetch(`${strapiBase}/api/sync/products`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        medusa_id: id,
        title: product.title,
        description: product.description,
        thumbnail: product.thumbnail,
        handle: product.handle,
        status: product.status,
        updated_at: new Date().toISOString()
      })
    })
  }

  if (action === 'deleted') {
    await fetch(`${strapiBase}/api/sync/products/${id}`, {
      method: 'DELETE'
    })
  }
}

export const config: SubscriberConfig = {
  event: ['product.created', 'product.updated', 'product.deleted'],

## 五、集成架构（核心）

### 5.1 Strapi ↔ Medusa 同步机制

```
┌──────────────┐      事件       ┌──────────────┐
│   Strapi     │ ─────────────→ │   Medusa     │
│              │                 │              │
│ entry.create │ Webhook POST    │ Subscriber   │
│ entry.update │ /webhooks/strapi│ 接收处理      │
│ entry.publish│                │              │
└──────────────┘                └──────────────┘
         ↑                              ↑
         │     Webhook POST              │
         └───────────────────────────────┘
              (产品/用户同步回调)
```

**Strapi Webhook 配置**：

```
Strapi Admin → Settings → Webhooks → Create Webhook

名称: medusa-sync
URL: https://api.your-site.com/webhooks/strapi
Headers: x-webhook-secret: your-secret
Events:
  ✓ entry.create (articles, videos, landing-pages)
  ✓ entry.update
  ✓ entry.publish / entry.unpublish
  ✓ media.create / media.update
```

**Medusa Subscriber 处理 Webhook**：

```typescript
// medusa/src/api/webhooks/strapi/route.ts
// 接收 Strapi Webhook，路由: POST /webhooks/strapi
import { MedusaAuthenticatedTaskRouting交通 } from '@medusajs/framework'

export const POST = async (req: MedusaRequest, res: MedusaResponse) => {
  const secret = req.headers['x-webhook-secret']
  if (secret !== process.env.WEBHOOK_SECRET) {
    return res.status(401).json({ error: 'Unauthorized' })
  }

  const { event, model, entry } = req.body

  if (model === 'article') {
    // 刷新 Next.js ISR 缓存
    await fetch(`${NEXT_SITE_URL}/api/revalidate?path=/blog/${entry.slug}`)
  }

  if (model === 'video') {
    // 更新视频 CDN 缓存
    await fetch(`${NEXT_SITE_URL}/api/revalidate?path=/videos/${entry.slug}`)
  }

  if (model === 'landing-page') {
    // 刷新 Landing Page
    await fetch(`${NEXT_SITE_URL}/api/revalidate?path=/${entry.slug}`)
  }

  res.status(200).json({ received: true })
}
```

// middleware.ts - NextAuth.js + 自定义 JWT 验证
import { NextResponse } from 'next/server'

export async function middleware(request) {
  const token = request.cookies.get('strapi_token')?.value

  if (token) {
    // 验证 Strapi JWT
    const strapiUser = await verifyStrapiToken(token)

    // 生成 Medusa-compatible token（或者直接用 Strapi token）
    const medusaToken = await generateMedusaToken(strapiUser)

    const response = NextResponse.next()
    response.cookies.set('strapi_token', token, { httpOnly: true })
    response.cookies.set('medusa_customer_id', strapiUser.medusa_customer_id)

    return response
  }

  return NextResponse.redirect('/login')
}

export const config = {
  matcher: ['/account/:path*', '/checkout/:path*', '/orders/:path*']
}
